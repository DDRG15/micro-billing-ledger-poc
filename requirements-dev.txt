# requirements-dev.txt — Development toolchain for posthog-billing-poc
# ======================================================================
# Install with: pip install -r requirements-dev.txt
#     These tools are NOT needed to run the app or the test suite.
#     They are for code quality checks during development.
#
# Usage — the ruff check and bandit module lists match .github/workflows/lint.yml, so a local run and CI
#     disagree only when one of them is out of date. CI runs those two and nothing else from
#     this file; `ruff format` and `mypy` are tools to run on demand, not gates:
#   ruff check ledger.py worker.py provision.py send_test_webhook.py disposable_db.py test_ledger.py benchmark.py scripts/build_architecture_viewer.py e2e/check_run.py e2e/downstream.py e2e/loadgen.py e2e/stripe_sandbox.py  # lint
#   ruff format ledger.py                                                 # format
#   mypy ledger.py                                                        # type check
#   bandit -r ledger.py worker.py provision.py send_test_webhook.py disposable_db.py e2e/check_run.py e2e/downstream.py e2e/loadgen.py e2e/stripe_sandbox.py -ll  # security scan

# ruff — extremely fast Python linter and formatter (replaces flake8 + isort + pyupgrade).
#     Detects unused imports, undefined names, style violations, and many more.
ruff==0.4.4

# mypy — static type checker. Enforces that type annotations (like Optional[int])
#     are actually respected in the code. Catches type mismatches before runtime.
mypy==1.10.0

# bandit — security-focused linter. Scans for hardcoded passwords, unsafe calls,
#     SQL injection patterns, and other OWASP Top 10 vulnerabilities in Python code.
#     CI runs it with -ll (medium severity and above). The one SQL finding in ledger.py, on
#     the DLQ listing query, is a false positive and carries a justified `# nosec B608`.
#     docs/testing.md has the recipe that shows bandit still detects shell=True, eval and
#     concatenated SQL.
bandit==1.7.8
