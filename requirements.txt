# requirements.txt — Micro-Billing-Ledger PoC
# =============================================
# All production dependencies pinned to exact versions.
#     Pinning prevents "it worked yesterday" surprises when a new minor version
#     ships a breaking change. Update pins intentionally, not accidentally.
#     Install with: pip install -r requirements.txt
#

# FastAPI 0.115.14 — HTTP framework for the webhook endpoint and summary/DLQ routes.
#     Chosen over Flask because it ships async support, automatic OpenAPI docs, and
#     Pydantic v2 integration out of the box.
#
#     Was 0.115.0, which pins starlette<0.39.0. Starlette below 0.40.0 carries
#     GHSA-f96h-pmfr-66vw (CVE-2024-47874): unbounded multipart form parsing, a
#     denial of service. This project exposes no multipart endpoint, so real
#     exposure was low — but the vulnerable version sat in the dependency tree and
#     any scanner would flag it. Dependabot did not, because starlette is not in
#     this manifest; it arrives through FastAPI's pin range, which is exactly how a
#     transitive advisory goes unnoticed.
#     0.115.14 requires starlette>=0.40.0 and is the last patch of the same minor
#     line, so the upgrade carries no API change. Verified green at the time (213/213);
#     the suite is 337/337 today.
fastapi==0.115.14

# uvicorn[standard] 0.30.6 — ASGI server that runs the FastAPI app.
#     [standard] includes uvloop (faster event loop, C extension) and httptools
#     (faster HTTP parsing). Both require the gcc/musl-dev in the builder Dockerfile stage.
#     For production with PostgreSQL and multiple workers, consider Gunicorn + uvicorn workers.
uvicorn[standard]==0.30.6

# Pydantic 2.9.2 — the entire validation stack runs on this.
#     v2 (not v1) is required because we use model_dump() (not deprecated dict()),
#     model_validator(mode='after'), and Field constraints with ge=0/pattern/min_length.
#     DO NOT downgrade to v1 — the API is incompatible and the validators will silently fail.
pydantic==2.9.2

# ACTIVE DEPENDENCIES

# Stripe 10.12.0 — SDK for webhook signature verification.
#     stripe.Webhook.construct_event() validates the HMAC-SHA256 signature in the
#     Stripe-Signature header. Without this, anyone who knows the URL can POST fake events.
#     Set STRIPE_WEBHOOK_SECRET in the environment to activate enforcement.
stripe==10.12.0

# psycopg2-binary 2.9.12 — PostgreSQL sync driver. ACTIVE.
#     Binary package includes libpq; no OS headers needed.
#     Was pinned to 2.9.9, which has no cp313 wheel: anyone on Python 3.13 following
#     the README's own stated prerequisite hit "pg_config executable not found" as pip
#     fell back to a source build. 2.9.12 ships wheels for 3.12 and 3.13, so the pin
#     and the documented prerequisite finally agree.
psycopg2-binary==2.9.12

# httpx 0.27.2 — required by starlette.testclient (TestClient transport layer).
#     starlette>=0.21 dropped requests in favour of httpx. Without it, importing
#     TestClient raises RuntimeError before any test runs.
httpx==0.27.2

# slowapi 0.1.9 — rate limiting for FastAPI. Wraps the `limits` library.
#     Limits: 100 req/min on POST /webhook/stripe, 20 req/min on GET /dlq/entries
#     and GET /ledger/summary. Prevents connection pool exhaustion and DLQ flooding.
slowapi==0.1.9

# alembic 1.18.4 — database migration framework.
#     Replaces _bootstrap() CREATE TABLE IF NOT EXISTS for production schema management.
#     Migrations live in alembic/versions/. Run: alembic upgrade head
alembic==1.18.4

# SQLAlchemy 2.0.41 — required by alembic as its engine layer.
#     Only used for migration execution, not for ORM queries (psycopg2 handles all app queries).
sqlalchemy==2.0.41

# prometheus-client 0.21.0 — Prometheus exposition for GET /metrics.
#     Provides Counter/Gauge/Histogram and generate_latest(). Used instead of
#     hand-rolling the text format because the escaping rules for label values and
#     the histogram bucket layout are easy to get subtly wrong, and a metrics
#     endpoint that emits almost-valid exposition fails at the scraper, not here.
prometheus-client==0.21.0
